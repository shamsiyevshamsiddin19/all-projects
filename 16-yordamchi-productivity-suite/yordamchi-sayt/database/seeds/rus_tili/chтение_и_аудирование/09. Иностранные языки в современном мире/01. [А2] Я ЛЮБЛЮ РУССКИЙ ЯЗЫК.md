# [А2] Я ЛЮБЛЮ РУССКИЙ ЯЗЫК

{Одна́жды|bir kun}, {когда́|qachon} {я|men} {была|edi} {в|da / ga / ichida} {Ве́нгрии|vengriya}, {я|men} {познако́милась|uchrashdi} {с|bilan / dan} {одно́й|bitta} {знамени́той|mashhur} {же́нщиной|ayol}, {кото́рую|qaysi} {зову́т|chaqirishadi / ismi} {Като́|kato} {Ломб|lomb}.
:: Bir kuni men Vengriyada bo'lganimda Kato Lomb ismli mashhur ayolni uchratdim.
{Э́та|bu} {же́нщина|ayol} {рабо́тает|ishlaydi} {перево́дчицей|tarjimon}.
:: Bu ayol tarjimon bo'lib ishlaydi.
{Она́|u} {зна́ет|biladi} 16 {языко́в|tillar}.
:: U 16 ta tilni biladi.

– {Во́семь|sakkiz} {языко́в|tillar} {я|men} {зна́ю|bilaman} {акти́вно|faol}, {а|esa / lekin} {во́семь|sakkiz} {пасси́вно|passiv}, – {говори́т|gapiradi / deydi} {она́|u}.
:: "Men sakkizta tilni faol, sakkiztasini passiv bilaman", deydi u.
{Она́|u} {зна́ет|biladi} {акти́вно|faol} {францу́зский|frantsuz}, {англи́йский|ingliz}, {испа́нский|ispancha}, {неме́цкий|nemis}, {кита́йский|xitoy}, {русский|rus}, {япо́нский|yapon}, {а|esa / lekin} {пасси́вно|passiv} – {лати́нский|lotin}, {румы́нский|rumin}, {по́льский|polyak}, {болга́рский|bolgar} {и|va} {други́е|boshqalar} {языки́|tillar}.
:: U fransuz, ingliz, ispan, nemis, xitoy, rus, yapon tillarini faol biladi va passiv ravishda lotin, rumin, polyak, bolgar va boshqa tillarni biladi.

{Като́|kato} {изуча́ла|o'rgangan} {в|da / ga / ichida} {шко́ле|maktab} {два|ikki} {языка́|til}: {францу́зский|frantsuz} {и|va} {лати́нский|lotin}.
:: Kato maktabda ikki tilni o'rgangan: frantsuz va lotin.
{Кита́йский|xitoy} {язы́к|til} {она́|u} {изуча́ла|o'rgangan} {с|bilan / dan} {учи́тельницей|o'qituvchi}, {а|esa / lekin} {все́|hamma / barcha} {остальны́е|qolganlari} {языки́|tillar} {она́|u} {изуча́ла|o'rgangan} {самостоя́тельно|o'zingizga}.
:: U o'qituvchi bilan xitoy tilini o'rgangan va boshqa barcha tillarni mustaqil o'rgangan.

– {Мой|mening} {са́мый|eng ko'p} {люби́мый|sevimli} {язы́к|til} – {ру́сский|rus}, – {говори́т|gapiradi / deydi} {Като́|kato} {Ломб|lomb}.
:: "Mening sevimli tilim rus tili", deydi Kato Lomb.
{Э́то|bu} {язы́к|til}, {у|da / huzurida} {кото́рого|qaysi} {са́мая|eng ko'p} {тру́дная|qiyin} {структу́ра|tuzilishi}, {э́то|bu} {са́мый|eng ko'p} {краси́вый|go'zal} {язы́к|til}.
:: Bu eng qiyin tuzilishga ega bo'lgan til, eng go'zal tildir.

{Като́|kato} {говори́т|gapiradi / deydi} {по-ру́сски|ruscha / rus tilida} {пра́вильно|to'g'ri}, {без|siz} {оши́бок|xatolar}, {иногда́|ba'zan} {то́лько|faqat} {де́лает|qiladi} {непра́вильное|noto'g'ri} {ударе́ние|ta'kidlash}.
:: Kato rus tilida to'g'ri, xatosiz gapiradi, ba'zida u shunchaki noto'g'ri urg'u qo'yadi.

– {Я|men} {была́|edi} {во|da / ga} {мно́гих|ko'p} {стра́нах|mamlakatlar} {и|va} {встреча́лась|uchrashdi} {с|bilan / dan} {разли́чными|har xil} {людьми́|odamlar}.
:: – Ko‘p mamlakatlarda bo‘lganman, turli odamlar bilan tanishganman.
{Я|men} {ду́маю|menimcha}, {что|nima / deb} {сейча́с|hozir} {зна́ние|bilim} {языко́в|tillar} {как|qanday / kabi} {никогда́|hech qachon} {ва́жно|muhim}.
:: Menimcha, hozir tillarni bilish har qachongidan ham muhimroq.
{Мир|tinchlik} {на|da / ustida / ga} {земле́|yer} {не|emas / -ma} {мо́жет|oladi / mumkin} {быть|bo'lmoq} {без|siz} {понима́ния|tushunish} {друг|do'st} {дру́га|do'st}, {без|siz} {встреч|uchrashuvlar}, {без|siz} {понима́ния|tushunish} {и|va} {зна́ния|bilim} {разли́чных|har xil} {культу́р|ekinlar}.
:: Bir-birini tushunmasdan, uchrashuvlarsiz, turli madaniyatlarni tushunmasdan va bilmasdan turib, er yuzida tinchlik bo'lmaydi.
{Что́бы|uchun} {понима́ть|tushunish} {друг|do'st} {дру́га|do'st}, {мы|biz} {должны́|kerak} {знать|bilish} {языки́|tillar}.
:: Bir-birimizni tushunishimiz uchun tillarni bilishimiz kerak.

{Като́|kato} {Ломб|lomb} – {челове́к|odam}, {кото́рый|qaysi} {не|emas / -ma} {то́лько|faqat} {лю́бит|yaxshi ko'radi} {свою́|sizning} {рабо́ту|ish}, {но|lekin} {и|va} {увлека́ется|olib ketadi} {е́ю|u bilan}.
:: Kato Lomb nafaqat o'z ishini yaxshi ko'radigan, balki unga ishtiyoqli odam.
{Она́|u} {подари́ла|berdi} {мне|menga} {кни́гу|kitob}, {кото́рую|qaysi} {она́|u} {написа́ла|yozgan}.
:: U menga yozgan kitobini berdi.
{Кни́га|kitob} {называ́ется|chaqirdi} «{Как|qanday / kabi} {я|men} {вы́учила|o'rgangan} 16 {языко́в|tillar}».
:: Kitob "Men 16 ta tilni qanday o'rgandim" deb nomlangan.
{Като́|kato} {Ломб|lomb} {лю́бит|yaxshi ko'radi} {литерату́ру|adabiyot}, {мно́го|ko'p} {чита́ет|o'qiydi}, {занима́ется|shug'ullanadi} {гимна́стикой|gimnastika}.
:: Kato Lomb adabiyotni yaxshi ko'radi, ko'p o'qiydi va gimnastika bilan shug'ullanadi.
{Я|men} {спроси́л|so'radi} {Като́|kato}, {как|qanday / kabi} {она́|u} {изуча́ет|o'rganadi} {иностра́нные|xorijiy} {языки́|tillar}.
:: Men Katodan chet tillarini qanday o'rganishini so'radim.
{Она́|u} {сказа́ла|dedi}: – {Что́бы|uchun} {хорошо́|yaxshi} {знать|bilish} {язы́к|til}, {ну́жно|kerak} {занима́ться|shug'ullanish} {ка́ждый|har bir} {день|kun}.
:: U shunday dedi: "Tilni yaxshi bilish uchun har kuni o'qish kerak."
{Челове́к|odam} {до́лжен|kerak} {испо́льзовать|foydalanish} {всё|hammasi}, {что|nima / deb} {даёт|beradi} {совреме́нная|zamonaviy} {те́хника|texnologiya}: {фи́льмы|filmlar}, {му́зыку|musiqa}.
:: Inson zamonaviy texnologiya taqdim etadigan hamma narsadan foydalanishi kerak: filmlar, musiqa.
{На́до|zarur} {слу́шать|tingla}, {чита́ть|o'qimoq}, {говори́ть|gapirmoq}.
:: Siz tinglashingiz, o'qishingiz, gapirishingiz kerak.
{Что́бы|uchun} {изучи́ть|o'rganish} {язы́к|til}, {ну́жно|kerak} {то́лько|faqat} {большо́е|katta} {жела́ние|istak} {знать|bilish} {его́|uni / uning}.
:: Tilni o'rganish uchun faqat uni bilishga bo'lgan katta istak kerak.
{Я|men} {не|emas / -ma} {встреча́ла|uchrashdi} {ни|hech / na} {одного́|bitta} {челове́ка|odam}, {кото́рый|qaysi} {не|emas / -ma} {мог|mumkin} {вы́учить|o'rganish} {язы́к|til}.
:: Til o‘rgana olmagan birorta odamni uchratmadim.
